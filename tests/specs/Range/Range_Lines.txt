== should format only the touched instruction ==
WORKDIR   /before
RUN   echo   [|hi|]
WORKDIR   /after

[expect]
WORKDIR   /before
[|RUN echo hi|]
WORKDIR   /after

WORKDIR   /after

[expect]
WORKDIR   /before
RUN echo hi
WORKDIR   /after
