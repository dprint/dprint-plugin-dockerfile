== should format a touched healthcheck instruction ==
WORKDIR   /before
HEALTHCHECK   --interval=30s    --timeout=3s   CMD   [|curl|]   -f   http://localhost/
WORKDIR   /after

[expect]
WORKDIR   /before
[|HEALTHCHECK --interval=30s --timeout=3s CMD curl -f http://localhost/|]
WORKDIR   /after

== should format a touched healthcheck instruction with an exec form ==
WORKDIR   /before
healthcheck   --retries=3   cmd   ["curl",[|"-f"|],"http://localhost/"]
WORKDIR   /after

[expect]
WORKDIR   /before
[|HEALTHCHECK --retries=3 CMD ["curl", "-f", "http://localhost/"]|]
WORKDIR   /after

== should keep the line continuation of a touched healthcheck instruction ==

== should move a comment out of a touched healthcheck instruction ==

#healthcheck comm[||]ent

[expect]

  CMD curl
WORKDIR   /after

== should keep the continuation inside the command of a touched healthcheck instruction ==

WORKDIR   /after

[expect]
WORKDIR   /before
[|HEALTHCHECK --interval=5m \
  CMD curl -f http://localhost/ \
    && echo ok|]
WORKDIR   /after
