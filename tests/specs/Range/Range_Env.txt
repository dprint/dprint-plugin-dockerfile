== should format a touched env instruction ==
WORKDIR   /before
ENV   MY_NAME=[|"John Doe"|]
WORKDIR   /after

[expect]
WORKDIR   /before
[|ENV MY_NAME="John Doe"|]
WORKDIR   /after

== should format a touched env instruction with several vars ==

env   MY_CAT=fluffy    [|Test|]=asdf
WORKDIR   /after

[expect]
WORKDIR   /before
[|ENV MY_CAT=fluffy Test=asdf|]
WORKDIR   /after

== should break a touched env instruction exceeding the line width ==
WORKDIR   /before
ENV   MY_NAME="John Doe"   [|MY_DOG|]=RexTheDog   MY_CAT=fluffy

[expect]
WORKDIR   /before
[|ENV MY_NAME="John Doe" \
    MY_DOG=RexTheDog \
    MY_CAT=fluffy|]
WORKDIR   /after
