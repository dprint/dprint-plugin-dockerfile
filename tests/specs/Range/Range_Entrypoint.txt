== should format a touched entrypoint instruction ==
WORKDIR   /before
ENTRYPOINT   [|echo|]   hi
WORKDIR   /after

WORKDIR   /before
[|ENTRYPOINT echo hi|]
WORKDIR   /after

[|ENTRYPOINT ["node", "main.js"]|]
