== should indent the instructions in a stage ==
FROM node AS base
WORKDIR /app

FROM base AS runner
USER node
EXPOSE 8000
CMD ["node", "server.js"]

[expect]
FROM node AS base
  WORKDIR /app

FROM base AS runner
  USER node
  EXPOSE 8000
  CMD ["node", "server.js"]

== should keep the instructions before the first stage at the outer level ==
# syntax=docker/dockerfile:1
ARG NODE_VERSION=20

FROM node:${NODE_VERSION}
RUN echo hi

[expect]
# syntax=docker/dockerfile:1
ARG NODE_VERSION=20

FROM node:${NODE_VERSION}
  RUN echo hi

== should re-indent an over indented stage ==

        WORKDIR /app
        RUN echo hi

FROM node AS base
  WORKDIR /app
  RUN echo hi

[expect]
FROM alpine
  HEALTHCHECK --interval=30s --timeout=3s --start-period=40s --retries=3 \
    CMD curl --fail --silent http://localhost:8080/healthz || exit 1
